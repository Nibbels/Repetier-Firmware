### Reproduction

Run with Arduino AVR Boards 1.8.6 and its AVR tools on the path.

```sh
python3 tests/communication-format.py
python3 tests/communication-format.py --ubsan
python3 tests/serial-protocol.py

out=$(mktemp -d)
for variant in RF1000 RF2000 RF2000v2 RF2000-X44; do
    flags="-DMOTHERBOARD=DEVICE_TYPE_${variant%-X44}"
    if [ "$variant" = RF2000-X44 ]; then
        flags="$flags -DNUM_EXTRUDER=1 -DEXT0_EXTRUDER_COOLER_PIN=X44_PIN"
    fi
    mkdir "$out/$variant"
    arduino-cli compile --fqbn arduino:avr:mega:cpu=atmega2560 --jobs 3 \
        --build-path "$out/$variant/build" \
        --build-property "compiler.cpp.extra_flags=$flags" \
        --build-property "compiler.c.elf.extra_flags=-Wl,-Map,$out/$variant/firmware.map,--cref" \
        --verbose Repetier
    avr-size -A "$out/$variant/build/Repetier.ino.elf"
    avr-objdump -d -C "$out/$variant/build/Repetier.ino.elf" > "$out/$variant/disassembly.txt"
done
```
